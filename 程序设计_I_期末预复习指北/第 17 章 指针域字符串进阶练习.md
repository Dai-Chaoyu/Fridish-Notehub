3. 杨辉三角

    ```c
    // 输出n行杨辉三角的函数
    void printYangHuiTriangle(int n) {
        // 边界条件检查：n必须是正整数
        if (n <= 0) {
            printf("请输入大于0的正整数！\n");
            return;
        }
    
        // 方法1：使用变量长度数组（C99及以上支持，简单直观）
        int triangle[n][n];  // 二维数组存储杨辉三角，实际只用左下三角部分
    
        // 填充杨辉三角
        for (int i = 0; i < n; i++) {  // i表示行号（从0开始）
            for (int j = 0; j <= i; j++) {  // j表示列号（从0开始），每行只填到第i列
                // 每行第一个或最后一个元素设为1
                if (j == 0 || j == i) {
                    triangle[i][j] = 1;
                } else {
                    // 中间元素 = 上一行前一列 + 上一行当前列
                    triangle[i][j] = triangle[i-1][j-1] + triangle[i-1][j];
                }
            }
        }
    
        // 输出杨辉三角（格式对齐，每个数占4个字符宽度）
        printf("杨辉三角（%d行）：\n", n);
        for (int i = 0; i < n; i++) {
            // 打印每行的元素
            for (int j = 0; j <= i; j++) {
                printf("%4d", triangle[i][j]);  // %4d 保证每个数占4格，对齐更美观
            }
            printf("\n");  // 每行结束换行
        }
    }
    ```

5. 螺旋矩阵

    ```c
    /**
     * Return an array of arrays of size *returnSize.
     * The sizes of the arrays are returned as *returnColumnSizes array.
     * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
     */
    int** generateMatrix(int n, int* returnSize, int** returnColumnSizes) {
        // 1. 设置返回的矩阵行数（即n）
        *returnSize = n;
        
        // 2. 分配列数数组的内存，并设置每行列数为n（因为是n×n矩阵）
        *returnColumnSizes = (int*)malloc(n * sizeof(int));
        for (int i = 0; i < n; i++) {
            (*returnColumnSizes)[i] = n;
        }
        
        // 3. 分配n×n矩阵的内存
        int** matrix = (int**)malloc(n * sizeof(int*));
        for (int i = 0; i < n; i++) {
            matrix[i] = (int*)malloc(n * sizeof(int));
        }
        
        // 4. 初始化边界与计数变量
        int top = 0;       // 上边界（初始为第0行）
        int bottom = n - 1;// 下边界（初始为第n-1行）
        int left = 0;      // 左边界（初始为第0列）
        int right = n - 1; // 右边界（初始为第n-1列）
        int count = 1;     // 待填充的数字（从1开始）
        
        // 5. 按顺时针方向填充矩阵
        while (count <= n * n) {
            // 步骤1：从左到右填充「上边界」行
            for (int i = left; i <= right; i++) {
                matrix[top][i] = count++;
            }
            top++; // 上边界下移
            
            // 步骤2：从上到下填充「右边界」列
            for (int i = top; i <= bottom; i++) {
                matrix[i][right] = count++;
            }
            right--; // 右边界左移
            
            // 步骤3：从右到左填充「下边界」行（需判断边界是否重叠，避免重复填充）
            if (top <= bottom) {
                for (int i = right; i >= left; i--) {
                    matrix[bottom][i] = count++;
                }
                bottom--; // 下边界上移
            }
            
            // 步骤4：从下到上填充「左边界」列（需判断边界是否重叠，避免重复填充）
            if (left <= right) {
                for (int i = bottom; i >= top; i--) {
                    matrix[i][left] = count++;
                }
                left++; // 左边界右移
            }
        }
        
        return matrix;
    }
    ```
